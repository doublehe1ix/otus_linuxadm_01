# Занятие 1. Обновление ядра системы

# Цель домашнего задания
# Научиться обновлять ядро в ОС Linux.
# Описание домашнего задания
# 1) Запустить ВМ c Ubuntu.
# 2) Обновить ядро ОС на новейшую стабильную версию из mainline-репозитория.
# 3) Оформить отчет в README-файле в GitHub-репозитории.

# Дополнительное задание:
# Собрать ядро самостоятельно из исходных кодов.

# Переключаемся на сессию суперпользователя, чтобы не делать каждый раз sudo
sudo su
# Устанавливаем необходимые зависимости
apt install -y build-essential libncurses-dev bison flex libssl-dev libelf-dev
# Загружаем самое свежее стабильное ядро 
wget https://cdn.kernel.org/pub/linux/kernel/v6.x/linux-6.17.7.tar.xz
# Распаковываем архив с ядром
tar -xf linux-6.17.7.tar.xz
# Переходим в директорию с распакованным ядром
cd linux-6.17.7/
# Копируем содержимое текущего конфига ядра в файл конфига для нового ядра
cp /boot/config-$(uname -r) .config
# Принимаем значения по умолчанию для всех параметров, которых не было в прежней версии ядра
make olddefconfig
# Убираем добавление дополнительных сертификатов из старого конфига
scripts/config --set-str SYSTEM_TRUSTED_KEYS ""
scripts/config --set-str SYSTEM_REVOCATION_KEYS ""
# Запускаем сборку с количеством потоков, равным числу доступных CPU
make -j$(nproc)
# Устанавливаем модули ядра
make modules_install
# Устанавливаем ядро
make install
# Перезагружаем машину для применения изменений
reboot
# Убеждаемся в том, что загрузилось ядро нужной версии (ожидаемый вывод 6.17.7)
uname -r
# Замораживаем обновления ядра с помощью менеджера пакетов из репозитория
sudo apt-mark hold linux-image*
